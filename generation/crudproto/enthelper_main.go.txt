package main

import (
	"context"
	"io"
	"os"

	"github.com/nxnminieye/nexa/generation/internal/enthelper"
	"github.com/nxnminieye/nexa/generation/toolchain"
)

func main() {
	stdin, err := io.ReadAll(io.LimitReader(os.Stdin, toolchain.MaxStdinBytes+1))
	if err != nil || len(stdin) > toolchain.MaxStdinBytes {
		os.Exit(1)
	}
	stdout, err := enthelper.Execute(context.Background(), stdin)
	if err != nil {
		os.Exit(1)
	}
	if _, err := os.Stdout.Write(stdout); err != nil {
		os.Exit(1)
	}
}
